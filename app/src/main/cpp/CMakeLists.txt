cmake_minimum_required(VERSION 3.22.1)
project(zenzjni C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
# the phone build targets a fixed ABI; a PC build (tools/loop/build_zenz_host.sh) uses the host CPU
if (ANDROID)
    set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
endif()
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)

add_subdirectory(llama.cpp)

add_library(zenzjni SHARED zenz_jni.cpp)
target_compile_options(zenzjni PRIVATE -O3 -Wno-deprecated-declarations)
if (ANDROID)
    target_link_libraries(zenzjni llama ggml android log)
else()
    find_package(JNI REQUIRED)
    target_include_directories(zenzjni PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(zenzjni llama ggml)
endif()
